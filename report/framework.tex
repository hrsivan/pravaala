\section{Conventions}
%=========================================================================
\begin{itemize}[nosep]
\item Sets are calligraphic ($\Occ,\Voc,\B$); vectors bold lower-case
  ($\v{w}$); matrices bold upper-case ($\v{M}$); random variables upright
  capitals ($W, C, K, G$).
\item $\log$ is base~2 (bits) in all information-theoretic quantities; $\ln$
  is used where the base matters for an identity (PMI and SGNS).
  $0\log 0 := 0$.
\item $[n] = \{1,\dots,n\}$. For a finite multiset/sequence $S$ and a type $t$,
  $\cnt_S(t)$ is the number of occurrences of $t$ in $S$.
\item Sanskrit strings are written in IAST in prose and are SLP1 in code; the
  transliteration maps are fixed in \S\ref{sec:translit}.
\item Every empirical quantity is reported with its \emph{support} (the number
  of items on which it is computed) and, where meaningful, an interval or a
  permutation $p$-value.
\end{itemize}

\section{Linguistic and philosophical motivation}
%=========================================================================
\paragraph{Distributional structure.} Harris (1954) and Firth (1957) state the
distributional hypothesis: words with similar distributions over contexts have
similar meanings. The Indian tradition has a sharper version of the
\emph{inverse} problem, i.e.\ how context selects one meaning of an
ambiguous word. Bhartṛhari (\emph{Vākyapadīya}~II) lists the factors that fix
the intended sense (\emph{viśeṣa-smṛti-hetu}) of a word whose meaning is
otherwise undetermined: association (\emph{saṃsarga}), dissociation
(\emph{viprayoga}), co-presence (\emph{sāhacarya}), opposition
(\emph{virodhitā}), purpose (\emph{artha}), topic (\emph{prakaraṇa}), sign
(\emph{liṅga}), proximity of another word (\emph{śabdasyānyasya
saṃnidhi}), capacity (\emph{sāmarthya}), propriety (\emph{aucitī}), place,
time, gender and accent.\footnote{\todo{Verify verse numbers against Iyer's
translation; numbering differs between the Rau and Iyer editions
(approx.\ II.314--317).}} Several of these are exactly what a window-based
co-occurrence model can observe (\emph{saṃsarga}, \emph{sāhacarya},
\emph{saṃnidhi}); others (\emph{prakaraṇa}, \emph{deśa}, \emph{kāla}) are
document- or genre-level and motivate the text metadata we keep for every
sentence. This gives RQ3 a classical framing: which \emph{viśeṣa-smṛti-hetu}
are recoverable from distributional evidence?

\paragraph{Lexicography.} The Amarakośa groups synonyms
(\emph{paryāya}) into synsets and synsets into \emph{vargas}; the third
\emph{kāṇḍa} adds adjectives, a miscellany, homonyms
(\emph{nānārtha-varga}) and indeclinables. A varga is therefore an ontological
class (heaven, water, mankind, \dots), not a distributional one, which is
precisely why comparing the two partitions (RQ2) is informative.

\paragraph{Morphology.} A Sanskrit running text realises each word through
(i) \emph{sandhi}, i.e.\ phonological fusion across word boundaries, (ii)
nominal compounding (\emph{samāsa}), and (iii) inflection (\emph{vibhakti},
\emph{tiṅ}). Each multiplies the number of surface types and therefore the
sparsity a distributional model faces. Our views (\S\ref{sec:views}) undo these
one at a time.

\section{Data, views and partitions}\label{sec:views}

\subsection{Transliteration}\label{sec:translit}
Let $\Sigma$ be the Sanskrit phoneme inventory. A transliteration scheme $S$ is
an injective map $\mathrm{enc}_S:\Sigma\to\Gamma_S^{+}$, extended to strings
homomorphically. SLP1 and WX are letter-to-letter
($|\mathrm{enc}(\sigma)|=1$), so $\mathrm{enc}_{\text{SLP1}}\circ
\mathrm{enc}_{\text{WX}}^{-1}$ is a bijection of letter sets and the
conversion is exact. IAST is not prefix-free (\emph{a} is a prefix of
\emph{ai}); decoding uses greedy longest match, with the diaeresis (\emph{aï})
marking hiatus.

\begin{remark}[An encoding mismatch in EvalSan]\label{rem:wx}
EvalSan's relatedness and synonym files are WX-encoded, while its
categorisation/analogy files and released vocabulary are SLP1; no
conversion is applied. Only $34.4\%$ of the relatedness test words are spelled
identically in both schemes. We therefore treat the scheme as a factor in the
OOV decomposition (Def.~\ref{def:oov}).

\end{remark}

\subsection{Corpus and views}
Let $\Occ$ be the set of word occurrences in the DCS ($|\Occ| = 5{,}801{,}761$
in 770,945 sentences). Each $o\in\Occ$ carries annotations
$(\text{form}, \text{unsandhied}, \text{lemma}, \text{lemmaID}, \text{sense}^?)$.

\paragraph{Corpus subsets.} DCS assigns every chapter a time slot
(\texttt{lookup/chapter-info.xml}): slot 1 contains the Vedic texts (Saṃhitās,
Brāhmaṇas, Sūtras), slot 2 the epics and early classical works (Mahābhārata,
Rāmāyaṇa, Caraka, Aṣṭādhyāyī), and slots 3--5 the later classical,
Purāṇic, medieval and commentarial literature. We define
$\Occ^{\text{all}} = \Occ^{\text{ved}}\sqcup\Occ^{\text{cl}}\sqcup\Occ^{\varnothing}$
with $|\Occ^{\text{ved}}|=1{,}310{,}123$ (slot 1),
$|\Occ^{\text{cl}}|=4{,}490{,}448$ (slots 2--5) and 1,190 undated occurrences.
Every quantity below is indexed by the subset when it matters; we write
$\Occ$ for the subset in use.

\begin{definition}[View]
A \emph{view} is a map $\View_k:\Occ\to\Voc_k^*$ (to a token, or to the
empty string). We use
\[
\begin{array}{lll}
\vk{surf} & \text{sandhied strings} & \text{(orthographic tokens)}\\
\vk{seg} & \text{sandhi-resolved inflected forms} & (-\text{sandhi})\\
\vk{lem} & \text{lemma strings} & (-\text{inflection})\\
\vk{lid} & \text{lemma IDs} & (+\text{homograph split})\\
\vk{sense} & \text{sense ID where annotated, else \vk{lid}} & (+\text{sense split}).
\end{array}
\]
\end{definition}
\noindent Each view induces a partition $\Part_k = \{\View_k^{-1}(t) : t\in
\Voc_k\}$ of $\Occ$ (for \vk{surf} after the many-to-one alignment of words to
sandhi strings). Write $\Part\preceq\Part'$ if $\Part$ is coarser than $\Part'$.

\begin{proposition}\label{prop:chain}
$\Part_{\vk{lem}}\preceq\Part_{\vk{lid}}\preceq\Part_{\vk{sense}}$, whereas
$\Part_{\vk{seg}}$ and $\Part_{\vk{lem}}$ are in general incomparable.
\end{proposition}
\begin{proof}
$\View_{\vk{lem}}=\strip\circ\View_{\vk{lid}}$ and
$\View_{\vk{lid}}=\strip'\circ\View_{\vk{sense}}$, so each block of the finer
partition lies inside a block of the coarser one. For incomparability it
suffices to exhibit one form realising two lemmas (syncretism across
paradigms) and one lemma realised by two forms (any paradigm).
\end{proof}

Treat a uniformly random occurrence $o\sim\mathrm{Unif}(\Occ)$ and the random
variables $X_k = \View_k(o)$. By Prop.~\ref{prop:chain} and the chain rule,
\begin{equation}\label{eq:chain}
H(X_{\vk{lid}}) = H(X_{\vk{lem}}) + H(X_{\vk{lid}}\mid X_{\vk{lem}}),
\end{equation}
so $H(X_{\vk{lid}}\mid X_{\vk{lem}})$ is exactly the information (in bits per
token) that ID-lemmatisation adds over string-lemmatisation, i.e.\ the
homography it resolves. Likewise $H(X_{\vk{lem}}\mid X_{\vk{seg}})$ measures
form-level ambiguity resolved by gold lemmatisation and
$H(X_{\vk{seg}}\mid X_{\vk{lem}})$ the inflectional information it discards.

\section{Out-of-vocabulary accounting (RQ3)}
%=========================================================================
\begin{definition}[OOV rate]\label{def:oov}
For a benchmark word set $\B$, view $k$ and threshold $\tau\in\mathbb{N}$, let
$\Voc_k(\tau) = \{t : \cnt_k(t)\ge\tau\}$ and, for ID views,
$\strip(\Voc_k(\tau))$. Then
$\OOV_k(\tau) = |\B\setminus\Voc_k(\tau)|/|\B|$.
\end{definition}

\noindent With $\B^{\mathrm{raw}}$ the benchmark in its original encoding,
\begin{align*}
\OOV^{\mathrm{raw}}_{\vk{surf}} ={}&
\underbrace{(\OOV^{\mathrm{raw}}_{\vk{surf}}-\OOV_{\vk{surf}})}_{\text{encoding}}
+\underbrace{(\OOV_{\vk{surf}}-\OOV_{\vk{seg}})}_{\text{sandhi}}
+\underbrace{(\OOV_{\vk{seg}}-\OOV_{\vk{lem}})}_{\text{inflection}}\\
&+\underbrace{(\OOV_{\vk{lem}}-\OOV_{\vk{lid}})}_{\text{homograph split}}
+\underbrace{\OOV_{\vk{lid}}}_{\text{rare}\;+\;\text{absent}}.
\end{align*}

\begin{proposition}
Under the membership rule ``a string is in-vocabulary if some type realising
it has count $\ge\tau$'', the homograph-split term is $\le 0$.
\end{proposition}
\begin{proof}
For every string $s$, $\max_{t:\strip(t)=s}\cnt_{\vk{lid}}(t)\le
\sum_{t:\strip(t)=s}\cnt_{\vk{lid}}(t)=\cnt_{\vk{lem}}(s)$. Hence
$\strip(\Voc_{\vk{lid}}(\tau))\subseteq\Voc_{\vk{lem}}(\tau)$.
\end{proof}

\begin{remark}
The ``absent'' residual is not all lexical sparsity: about half of the absent
Amarakośa words are compounds or primary derivatives (e.g.\ participles), which
DCS analyses into members or roots, so they never surface as lemmas. Nasal
orthography (anusvāra vs.\ homorganic nasal) accounts for a further $\approx
1\%$. Resolving this is a design decision (see \S\ref{sec:decisions}).
\end{remark}

\section{Distributional models}
%=========================================================================
\subsection{Co-occurrence and pointwise mutual information}
Fix a view, its vocabulary $\Voc$ ($|\Voc|=n$) and a symmetric window $L$.
The co-occurrence matrix $\v{C}\in\R^{n\times n}_{\ge0}$ is
$C_{wc}=\sum_{(o,o')} a(d(o,o'))\,[\View(o)=w,\View(o')=c]$ over ordered pairs
within a sentence at distance $1\le d\le L$, with $a(d)=1/d$ (harmonic) or 1.
Normalising gives the empirical joint $\hat P(w,c)$ and marginals.

\begin{definition}[Smoothed, shifted PPMI]
For $\alpha\in(0,1]$ and $k\ge1$, with $P_\alpha(c)\propto\cnt(c)^\alpha$,
\[
\PMI_\alpha(w,c)=\ln\frac{\hat P(w,c)}{\hat P(w)P_\alpha(c)},\qquad
\SPPMI_{\alpha,k}(w,c)=\max\bigl(\PMI_\alpha(w,c)-\ln k,\,0\bigr).
\]
\end{definition}
Since $\E_{c\sim\hat P(\cdot\mid w)}[\PMI(w,c)] = D_{\mathrm{KL}}(\hat P(\cdot\mid
w)\,\|\,\hat P(\cdot))$, a word's PMI row is a decomposition of how far its
context distribution is from the background: frequent, unselective words
(particles, pronouns) have small divergence and low PMI everywhere. \emph{This
is the precise sense in which PMI ``reduces the effect of high-frequency
words''.}

\begin{proposition}[Levy \& Goldberg 2014]
If the SGNS objective with $k$ negative samples drawn from $P_\alpha$ is
optimised independently for every $(w,c)$ with unconstrained dimensionality,
its optimum satisfies $\v{w}^\top\v{c} = \PMI_\alpha(w,c)-\ln k$.
\end{proposition}
\noindent Hence count-based (SPPMI + SVD) and predictive (SGNS) models are two
estimators of a low-rank approximation of the same information-theoretic
object. We run both; differences between them are differences of
\emph{estimation}, not of target.

\begin{proposition}[GloVe as weighted PMI factorisation]
GloVe minimises $\sum_{ij:X_{ij}>0} f(X_{ij})\,(\v{w}_i^\top\v{u}_j+b_i+c_j-\ln
X_{ij})^2$ with $f(x)=\min(1,(x/x_{\max})^{3/4})$. If the biases equal
$b_i=\ln X_{i\cdot}$ and $c_j=\ln X_{\cdot j}-\ln N$, a zero residual means
$\v{w}_i^\top\v{u}_j=\PMI(i,j)$.
\end{proposition}
\noindent So GloVe is a weighted least-squares factorisation of a PMI-like
matrix, with $f$ down-weighting rare and noisy cells, while SPPMI-SVD is an
unweighted factorisation of the truncated, shifted PMI and SGNS an implicit
one. FastText is SGNS with $\v{w}$ replaced by a sum of character $n$-gram
vectors, which lets it produce vectors for unseen strings. All four models of
the study therefore estimate the same information-theoretic object under
different loss functions, weightings and parametrisations.

\subsection{Low-rank factorisation}
Let $\v{M}=\SPPMI(\v{C})$ and $\v{M}=\v{U}\v{\Sigma}\v{V}^\top$ its SVD. By the
Eckart--Young--Mirsky theorem,
$\v{U}_d\v{\Sigma}_d\v{V}_d^\top=\arg\min_{\operatorname{rk}\v{X}\le d}\|\v{M}-\v{X}\|_F$.
Word vectors are $\v{W}=\v{U}_d\v{\Sigma}_d^{\,p}$, $p\in\{0,\tfrac12,1\}$.
For $p=1$, $\v{W}\v{W}^\top$ is the best rank-$d$ approximation of
$\v{M}\v{M}^\top$ (second-order similarity).

\subsection{Frequency damping as surprisal}
word2vec subsampling keeps an occurrence of $w$ with probability
$\kappa(w)=\min\bigl(1,\sqrt{t/f(w)}\bigr)$. With self-information
$I(w)=-\log f(w)$ and $I_t=-\log t$,
\[
\kappa(w)=\min\bigl(1,\;2^{-(I_t-I(w))/2}\bigr),
\]
so occurrences carrying fewer than $I_t$ bits are discarded at a rate
exponential in their information deficit. In the count model we use the
\emph{expected} counts $\E[C_{wc}]=\kappa(w)\kappa(c)C_{wc}$, removing sampling
noise. Alternatives to compare: weighting by $I(w)$ (an IDF analogue) and by
the context entropy $H(C\mid W=w)$, which is frequency-independent.

\section{Evaluation}
%=========================================================================
\subsection{Similarity on a common scale}
Cosine values are not comparable across spaces (anisotropy, dimension).
All similarity tasks are therefore scored through rank statistics, which
are invariant under strictly monotone transformations of the scale:
Spearman $\rho$ between $\cos$ and the ordinal label
$y\in\{0,1,2\}$ (unrelated, same varga, same synset), and
\[
\mathrm{AUC}_{ij}=\Pr\bigl(s(X_i)>s(X_j)\bigr)+\tfrac12\Pr\bigl(s(X_i)=s(X_j)\bigr),
\]
the normalised Mann--Whitney $U$, for each label pair $i>j$. For ID views a
string $s$ may denote several types; we use MaxSim,
$\mathrm{sim}(s,s')=\max_{t\in s,\,t'\in s'}\cos(\v{t},\v{t}')$.
OOV items are excluded and coverage is always reported. EvalSan
\emph{imputes} the mean vector instead, which assigns cosine~1 (hence the
``same synset'' label) to every pair of two OOV words.

\paragraph{Common support.} Coverage is not neutral: a model with a higher
frequency threshold covers only frequent words, whose vectors are better
estimated, and so scores higher on a smaller, easier set. Comparisons between
models or views are therefore made on the \emph{common support}
$\B_\cap=\bigcap_m \B^{\mathrm{cov}}_m$, with paired tests on that set;
per-model full-coverage numbers are reported separately.

\paragraph{Two protocols.} For comparability with published numbers we also
run an \emph{EvalSan-faithful} protocol that replicates the original code task
by task. Relatedness imputes the mean vector for OOV words and uses fixed
thresholds $(0.25,0.5)$ with no tuning; the synonym MCQ \emph{skips} items
containing an OOV word; categorisation imputes the mean vector and reports the
\emph{maximum} purity over six clustering configurations, selected on the
test set itself, which is an optimistic bias. Its variant
\texttt{evalsan\_raw} also keeps the unconverted WX words, which isolates the
encoding effect of Remark~\ref{rem:wx}.

\subsection{Comparing partitions}
Let $K$ (cluster) and $G$ (gold class) be the labels of a uniformly random
evaluated word.
\begin{definition}
$\pur(K,G)=\sum_{k}\max_g\Pr(K=k,G=g)$;\quad
$\NMI(K,G)=\dfrac{2I(K;G)}{H(K)+H(G)}$;\quad
$\VI(K,G)=H(K\mid G)+H(G\mid K)$.
\end{definition}
\begin{proposition}
If $\mathcal{K}'$ refines $\mathcal{K}$ then $\pur(K',G)\ge\pur(K,G)$; in
particular purity is maximised (=1) by singletons.
\end{proposition}
\begin{proof}
If cluster $k$ splits into $k_1,\dots,k_m$, then
$\sum_i\max_g\Pr(k_i,g)\ge\max_g\sum_i\Pr(k_i,g)=\max_g\Pr(k,g)$.
\end{proof}
\noindent Purity therefore cannot compare clusterings with different numbers
of clusters; VI can, as it is a metric on the lattice of partitions (Meilă
2007), bounded by $\log|\B|$. We report all three, each with a permutation
null (gold labels shuffled, cluster sizes fixed) and its $p$-value.

\subsection{Gold relations are not functions}
In the Amarakośa a word may belong to several synsets and vargas (polysemy,
the \emph{nānārtha-varga}). Gold is a relation $R\subseteq\B\times\mathcal{G}$.
Default protocol: restrict to $\B_1=\{b:|R(b)|=1\}$ outside the
\emph{nānārtha} and \emph{avyaya} vargas; the polysemous complement
$\B\setminus\B_1$ is precisely the test bed for the \vk{lid}/\vk{sense} views
(RQ3).

\subsection{Optional: composition for unattested compounds}\label{sec:compose}
For $b\in\B$ absent from $\Voc_{\vk{lem}}$, the optional plugin proposes a
segmentation $s=(m_1,\dots,m_r)$ into attested lemmas, undoing junction sandhi
and stem alternations, and selects the one of minimum description length
under the unigram lemma model:
\[
s^\star=\arg\min_{s}\;L(s),\qquad L(s)=\sum_{i=1}^{r}-\log p(m_i),\qquad
p(m)=\cnt(m)/N.
\]
Each additional member costs bits, so over-segmentation is penalised without
an ad hoc prior. The confidence is the margin $\Delta=L(s_2)-L(s^\star)$; with
a uniform prior over segmentations, the posterior odds of $s^\star$ over $s_2$
are $2^{\Delta}$. Low surprisal does not guarantee a correct etymology
(\emph{akṣīva} ``drunk'' is not \emph{akṣi}+\emph{iva}), so only splits a
human has approved are used by default. Composition is additive,
$\v{v}(b)=\sum_i\v{v}(m_i)/\|\sum_i\v{v}(m_i)\|$ (Mitchell \& Lapata 2010).
It is order-blind, cannot tell a \emph{tatpuruṣa} from a \emph{dvandva},
and so gives a lower bound on what composition can achieve. Results are kept
in a separate result file and never mixed with the main tables.

\section{Hypotheses}
%=========================================================================
\begin{hypothesis}[RQ3, OOV]\label{h:oov}
A substantial share of the published EvalSan OOV rate is attributable to
encoding mismatch and inflection/sandhi rather than to lexical sparsity;
quantified by the decomposition of \S4.
\end{hypothesis}
\begin{hypothesis}[RQ1]
At equal corpus and hyperparameters, \vk{lem} and \vk{lid} views improve
coverage-adjusted relatedness and synonym scores over \vk{seg} and
\vk{surf}; the gain over \vk{seg} is larger for the count-based model (whose
sparsity is worse) than for SGNS.
\end{hypothesis}
\begin{hypothesis}[RQ1/RQ3, homography]
On pairs involving DCS homographs, \vk{lid} with MaxSim outperforms \vk{lem};
on non-homographs the two are indistinguishable (paired test).
\end{hypothesis}
\begin{hypothesis}[RQ2]
Unsupervised clusters recover vargas significantly above the permutation
null, and concrete vargas (flora/fauna, water, body) are recovered better than
abstract ones (cognition, sound, adjectives); the varga-level mean context
entropy $\bar H(C\mid W)$ is higher for poorly recovered vargas.
\end{hypothesis}
\begin{hypothesis}[RQ2, divergence]
Systematic cross-varga clusters correspond to textual co-presence
(\emph{sāhacarya}), e.g.\ deities with their mounts/weapons, rather than to
taxonomic similarity (relatedness vs.\ similarity, cf.\ Agirre et al.\ 2009).
\end{hypothesis}
